\section{Laplace transforms}

\begin{enumerate}[label=(\arabic*)]

\item
\[
\mathcal{L}\{\cos^2(\omega t)\} 
= \mathcal{L}\!\left\{\tfrac12\bigl(1+\cos(2\omega t)\bigr)\right\}
= \tfrac12\bigl(\mathcal{L}\{1\}+\mathcal{L}\{\cos(2\omega t)\}\bigr)
= \tfrac12\left(\frac{1}{s}+\frac{s}{s^{2}+4\omega^{2}}\right).
\]

\item
\[
\mathcal{L}\{t^2-1\}
= \mathcal{L}\{t^2\}-\mathcal{L}\{1\}
= \frac{2}{s^{3}}-\frac{1}{s}.
\]

\item
\[
\mathcal{L}\{e^{-t}\cos^{2}t\}
= \mathcal{L}\!\left\{e^{-t}\cdot \tfrac12\bigl(1+\cos(2t)\bigr)\right\}
= \tfrac12\,\mathcal{L}\{e^{-t}\}+\tfrac12\,\mathcal{L}\{e^{-t}\cos(2t)\}
= \frac{1}{2(s+1)}+\frac{s+1}{2((s+1)^{2}+4)}.
\]

\end{enumerate}

\section{Inverse Laplace transforms}

\begin{enumerate}[label=(\arabic*)]

\item
\[
\begin{aligned}
\mathcal{L}^{-1}\!\left\{\frac{1}{(s-a)(s-b)}\right\}
&= \mathcal{L}^{-1}\!\left\{\frac{1}{a-b}\!\left(\frac{1}{s-a}-\frac{1}{s-b}\right)\right\} \\
&= \frac{1}{a-b}\,\mathcal{L}^{-1}\!\left\{\frac{1}{s-a}-\frac{1}{s-b}\right\} \\
&= \frac{1}{a-b}\Bigl(\mathcal{L}^{-1}\!\left\{\frac{1}{s-a}\right\}
     -\mathcal{L}^{-1}\!\left\{\frac{1}{s-b}\right\}\Bigr) \\
&= \frac{1}{a-b}\bigl(e^{at}-e^{bt}\bigr) \\
&= \frac{e^{at}-e^{bt}}{a-b}\qquad (a\neq b).
\end{aligned}
\]
\emph{If $a=b$ then }$\displaystyle \mathcal{L}^{-1}\!\left\{\frac{1}{(s-a)^2}\right\}=t e^{at}$

\item
\[
\begin{aligned}
\mathcal{L}^{-1}\!\left\{\frac{s^{2}}{1+s^{2}}\right\}
&= \mathcal{L}^{-1}\!\left\{\frac{s^{2}+1-1}{s^{2}+1}\right\} \\[2pt]
&= \mathcal{L}^{-1}\!\left\{1-\frac{1}{s^{2}+1}\right\} \\[2pt]
&= \mathcal{L}^{-1}\{1\}-\mathcal{L}^{-1}\!\left\{\frac{1}{s^{2}+1}\right\} \\[2pt]
&= \delta(t)-\sin t.
\end{aligned}
\]

\end{enumerate}

\section{Ordinary differential equations}
\begin{enumerate}[label=(\arabic*)]
\item
\[
\begin{aligned}
&y''-y'+2y=2e^{3t},\quad y(0)=0,\ y'(0)=0.\\[2pt]
&(s^2Y - sy(0) - y'(0)) - (sY - y(0)) + 2Y = \frac{2}{s-3} \\[2pt]
&(s^2 - s + 2)Y = \frac{2}{s-3}
\;\Rightarrow\;
Y=\frac{2}{(s-3)(s^2 - s + 2)}.\\
&Y=\frac{1}{4}\frac{1}{s-3}-\frac{1}{4}\frac{s+2}{s^2 - s + 2}
=\frac{1}{4}\frac{1}{s-3}-\frac{1}{4}\frac{s-\tfrac12}{(s-\tfrac12)^2+\tfrac{7}{4}}-\frac{5}{4}\frac{1}{(s-\tfrac12)^2+\tfrac{7}{4}}.\\
&\Rightarrow\;
y(t)=\frac{1}{4}e^{3t}-\frac{1}{4}e^{t/2}\cos\!\frac{\sqrt7 t}{2}
-\frac{5}{4\sqrt7}\,e^{t/2}\sin\!\frac{\sqrt7 t}{2}.
\end{aligned}
\]

\item
\[
\begin{aligned}
&y''+4y'+4y=t^3*e^{-2t},\quad y(0)=1,\ y'(0)=2.\\[2pt]
&(s^2Y-sy(0)-y'(0))+4(sY-y(0))+4Y=\frac{6}{s^4}\cdot\frac{1}{s+2} \\[2pt]
&(s+2)^2Y=(s+6)+\frac{6}{s^4(s+2)}
\;\Rightarrow\;
Y=\frac{s+6}{(s+2)^2}+\frac{6}{s^4(s+2)^3}.\\
&\frac{s+6}{(s+2)^2}=\frac{1}{s+2}+\frac{4}{(s+2)^2},\\[2pt]
&\frac{6}{s^4(s+2)^3}
=\frac{-\tfrac{15}{16}}{s}+\frac{\tfrac{9}{8}}{s^2}-\frac{\tfrac{9}{8}}{s^3}
+\frac{\tfrac{3}{4}}{s^4}
+\frac{\tfrac{15}{16}}{s+2}+\frac{\tfrac{3}{4}}{(s+2)^2}+\frac{\tfrac{3}{8}}{(s+2)^3}.\\
&\Rightarrow\;
\,y(t)=\frac{31}{16}e^{-2t}+\frac{19}{4}t e^{-2t}+\frac{3}{16}t^2 e^{-2t}
-\frac{15}{16}+\frac{9}{8}t-\frac{9}{16}t^2+\frac{1}{8}t^3
\end{aligned}
\]

\item
\[
\begin{aligned}
&y'''+3y''+3y'+y=6e^{-t},\quad y(0)=y'(0)=y''(0)=0.\\[2pt]
&(s^3Y-s^2y(0)-s y'(0)-y''(0))
+3(s^2Y-sy(0)-y'(0))
+3(sY-y(0))+Y=\frac{6}{s+1}\\[2pt]
&(s^3+3s^2+3s+1)Y=\frac{6}{s+1}
\;\Leftrightarrow\;
(s+1)^3 Y=\frac{6}{s+1}
\;\Rightarrow\;
Y=\frac{6}{(s+1)^4}.\\
&\Rightarrow\;
y(t)=\mathcal{L}^{-1}\!\left\{\frac{6}{(s+1)^4}\right\}
= t^{3}e^{-t}.
\end{aligned}
\]

\item
\[
\begin{aligned}
&y^{(4)}+2y''+y=\sin t,\quad y(0)=y'(0)=y''(0)=y^{(3)}(0)=0.\\
&(s^4Y+2s^2Y+Y)=\frac{1}{s^2+1}
\;\Rightarrow\;
(s^2+1)^2Y=\frac{1}{s^2+1}
\;\Rightarrow\;
Y=\frac{1}{(s^2+1)^3}.\\
&\text{Write } \frac{1}{(s^2+1)^3}
= A\frac{1}{s^2+1}
+ B\!\left(-\frac{d}{ds}\frac{s}{s^2+1}\right)
+ C\!\left(\frac{d^2}{ds^2}\frac{1}{s^2+1}\right),\\
&A=\tfrac{3}{8},\quad B=-\tfrac{3}{8},\quad C=-\tfrac{1}{8}\\
&\mathcal{L}^{-1}\!\left\{\frac{1}{(s^2+1)^3}\right\}
=A\sin t+B\,t\cos t+C\,t^2\sin t\\
&\Rightarrow\;
y(t)=\frac{1}{8}\Bigl(3\sin t-3t\cos t-t^2\sin t\Bigr).
\end{aligned}
\]

\end{enumerate}
