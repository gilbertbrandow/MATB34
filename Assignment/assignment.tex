\section{Laplace transforms}

\begin{enumerate}[label=(\arabic*)]

\item
\[
\mathcal{L}\{\cos^2(\omega t)\} 
= \mathcal{L}\!\left\{\tfrac12\bigl(1+\cos(2\omega t)\bigr)\right\}
= \tfrac12\bigl(\mathcal{L}\{1\}+\mathcal{L}\{\cos(2\omega t)\}\bigr)
= \tfrac12\left(\frac{1}{s}+\frac{s}{s^{2}+4\omega^{2}}\right).
\]

\item
\[
\mathcal{L}\{t^2-1\}
= \mathcal{L}\{t^2\}-\mathcal{L}\{1\}
= \frac{2}{s^{3}}-\frac{1}{s}.
\]

\item
\[
\mathcal{L}\{e^{-t}\cos^{2}t\}
= \mathcal{L}\!\left\{e^{-t}\cdot \tfrac12\bigl(1+\cos(2t)\bigr)\right\}
= \tfrac12\,\mathcal{L}\{e^{-t}\}+\tfrac12\,\mathcal{L}\{e^{-t}\cos(2t)\}
= \frac{1}{2(s+1)}+\frac{s+1}{2((s+1)^{2}+4)}.
\]

\end{enumerate}

\section{Inverse Laplace transforms}

\begin{enumerate}[label=(\arabic*)]

\item
\[
\begin{aligned}
\mathcal{L}^{-1}\!\left\{\frac{1}{(s-a)(s-b)}\right\}
&= \mathcal{L}^{-1}\!\left\{\frac{1}{a-b}\!\left(\frac{1}{s-a}-\frac{1}{s-b}\right)\right\} \\[4pt]
&= \frac{1}{a-b}\,\mathcal{L}^{-1}\!\left\{\frac{1}{s-a}-\frac{1}{s-b}\right\} \\
&= \frac{1}{a-b}\Bigl(\mathcal{L}^{-1}\!\left\{\frac{1}{s-a}\right\}
     -\mathcal{L}^{-1}\!\left\{\frac{1}{s-b}\right\}\Bigr) \\
&= \frac{1}{a-b}\bigl(e^{at}-e^{bt}\bigr) \\
&= \frac{e^{at}-e^{bt}}{a-b}\qquad (a\neq b).
\end{aligned}
\]
\emph{If $a=b$ then }$\displaystyle \mathcal{L}^{-1}\!\left\{\frac{1}{(s-a)^2}\right\}=t e^{at}$

\item
\[
\begin{aligned}
\mathcal{L}^{-1}\!\left\{\frac{s^{2}}{1+s^{2}}\right\}
&= \mathcal{L}^{-1}\!\left\{\frac{s^{2}+1-1}{s^{2}+1}\right\} \\[2pt]
&= \mathcal{L}^{-1}\!\left\{1-\frac{1}{s^{2}+1}\right\} \\[2pt]
&= \mathcal{L}^{-1}\{1\}-\mathcal{L}^{-1}\!\left\{\frac{1}{s^{2}+1}\right\} \\[2pt]
&= \delta(t)-\sin t.
\end{aligned}
\]

\end{enumerate}
